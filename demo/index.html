Usage Examples

// Basic usage import { MatchMediaScreen } from '@phucbm/match-media-screen'; new MatchMediaScreen({ object: { value: 'desktop', responsive: [ { breakpoint: 1024, settings: { value: 'tablet' } }, { breakpoint: 480, settings: { value: 'mobile' } } ] }, onMatched: (data) => { console.log('Breakpoint matched:', data); }, onUpdate: (data) => { console.log('Window resized:', data); } }); // Advanced configuration new MatchMediaScreen({ object: { columns: 4, spacing: '2rem', fontSize: '16px', responsive: [ { breakpoint: 1024, settings: { columns: 2, spacing: '1.5rem' } }, { breakpoint: 768, settings: { columns: 1, spacing: '1rem', fontSize: '14px' } } ] }, isInherit: true, debounce: 150, dev: true });
